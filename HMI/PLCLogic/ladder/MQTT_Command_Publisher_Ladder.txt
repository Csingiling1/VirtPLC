(*
   LADDER LOGIC - MQTT Command Publisher (Ladder Diagram Representation)
   Convert button presses from Ignition HMI to MQTT commands for Unreal Engine
*)

(* ====================================================================================
   NETWORK 1: Conveyor 1 Start/Stop Command Publisher
   ==================================================================================== *)
   
   [Conveyor1_Running_Input]──┬──[NOT Conveyor1_Running_Prev]────(MQTT_PUBLISH_C1_ON)
                               │
                               └──[/]───────────────────────────────────────────────────

   Explanation:
   - When Conveyor1_Running changes from OFF to ON (positive edge)
   - Publishes: Topic="ignition/command/conveyor1/running" Payload={"value":true}

(*  ====================================================================================
    NETWORK 2: Conveyor 1 OFF Command
    ==================================================================================== *)

   [/Conveyor1_Running_Input]─┬──[Conveyor1_Running_Prev]────(MQTT_PUBLISH_C1_OFF)
                               │
                               └──[/]──────────────────────────────────────────────────

   Explanation:
   - When Conveyor1_Running changes from ON to OFF (negative edge)
   - Publishes: Topic="ignition/command/conveyor1/running" Payload={"value":false}

(* ====================================================================================
   NETWORK 3: Conveyor 2 Start/Stop Command Publisher
   ==================================================================================== *)

   [Conveyor2_Running_Input]──┬──[NOT Conveyor2_Running_Prev]────(MQTT_PUBLISH_C2_ON)
                               │
                               └──[/]───────────────────────────────────────────────────

   [/Conveyor2_Running_Input]─┬──[Conveyor2_Running_Prev]────(MQTT_PUBLISH_C2_OFF)
                               │
                               └──[/]──────────────────────────────────────────────────

(* ====================================================================================
   NETWORK 4: Machine 1 Start/Stop Command Publisher
   ==================================================================================== *)

   [Machine1_Running_Input]───┬──[NOT Machine1_Running_Prev]────(MQTT_PUBLISH_M1_ON)
                               │
                               └──[/]───────────────────────────────────────────────────

   [/Machine1_Running_Input]──┬──[Machine1_Running_Prev]────(MQTT_PUBLISH_M1_OFF)
                               │
                               └──[/]──────────────────────────────────────────────────

(* ====================================================================================
   NETWORK 5: Update Previous States (Must run after publishing)
   ==================================================================================== *)

   [ALWAYS_TRUE]──────────────────────┬──(MOV Conveyor1_Running_Input → Conveyor1_Running_Prev)
                                       │
                                       ├──(MOV Conveyor2_Running_Input → Conveyor2_Running_Prev)
                                       │
                                       └──(MOV Machine1_Running_Input → Machine1_Running_Prev)

(* ====================================================================================
   MQTT_PUBLISH Coil Implementation
   
   Each coil triggers an MQTT publish with the following structure:
   
   MQTT_PUBLISH_C1_ON:
     Topic: "ignition/command/conveyor1/running"
     Payload: {"value":true,"timestamp":<current_time>,"source":"plc"}
     QoS: 1
     Retain: false
   
   MQTT_PUBLISH_C1_OFF:
     Topic: "ignition/command/conveyor1/running"
     Payload: {"value":false,"timestamp":<current_time>,"source":"plc"}
     QoS: 1
     Retain: false
   
   (Similar for C2_ON/OFF and M1_ON/OFF with respective topics)
   ==================================================================================== *)

(* ====================================================================================
   TAG MAPPING REFERENCE
   ==================================================================================== *)
   
   (* Inputs from Ignition Edge Tags *)
   %I0.0 : Conveyor1_Running_Input  (linked to [default]Conveyor1/Running)
   %I0.1 : Conveyor2_Running_Input  (linked to [default]Conveyor2/Running)
   %I0.2 : Machine1_Running_Input   (linked to [default]Machine1/Running)
   
   (* Memory Bits for State Tracking *)
   %M0.0 : Conveyor1_Running_Prev
   %M0.1 : Conveyor2_Running_Prev
   %M0.2 : Machine1_Running_Prev
   
   (* MQTT Status Outputs *)
   %Q0.0 : MQTT_Connected
   %Q0.1 : MQTT_Publish_Success
   %Q0.2 : MQTT_Publish_Error
